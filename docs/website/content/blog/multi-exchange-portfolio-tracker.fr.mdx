Si vous tradez sur plus d'un exchange, vous connaissez le rituel : se connecter ici, se connecter
là, additionner de tête, oublier la poussière qui traîne sur ce troisième compte. Tout l'intérêt
d'une API unifiée, c'est que « combien ai-je vraiment ? » devrait tenir en un script — des clés
en lecture seule, aucune permission d'ordre, rien à casser.

Le plan : récupérer les soldes de chaque exchange, les fusionner par devise, tout valoriser en USDT
à partir des tickers d'une seule plateforme, afficher un tableau et un total.

```javascript tab="JavaScript"
import ccxt from 'ccxt';

const ids = ['binance', 'kraken', 'bybit'];
const totals = {};

for (const id of ids) {
    const exchange = new ccxt[id]({
        apiKey: process.env[`${id.toUpperCase()}_APIKEY`],
        secret: process.env[`${id.toUpperCase()}_SECRET`],
    });
    const balance = await exchange.fetchBalance();
    for (const [currency, amount] of Object.entries(balance.total)) {
        if (amount > 0) {
            totals[currency] = (totals[currency] ?? 0) + amount;
        }
    }
    await exchange.close();
}

// tout valoriser en USDT à partir des tickers d'une seule plateforme
const pricing = new ccxt.binance();
const tickers = await pricing.fetchTickers();
let portfolio = 0;
for (const [currency, amount] of Object.entries(totals)) {
    const price = (currency === 'USDT') ? 1 : tickers[`${currency}/USDT`]?.last;
    if (!price) continue;                    // pas de marché USDT — voir les notes ci-dessous
    const value = amount * price;
    portfolio += value;
    console.log(`${currency.padEnd(8)} ${amount.toFixed(6).padStart(14)} ≈ ${value.toFixed(2).padStart(12)} USDT`);
}
console.log(`${'TOTAL'.padEnd(8)} ${' '.repeat(14)} ≈ ${portfolio.toFixed(2).padStart(12)} USDT`);
await pricing.close();
```

```go tab="Go"
package main

import (
    "fmt"
    "os"
    "sort"
    "strings"

    ccxt "github.com/ccxt/ccxt/go/v4"
)

func main() {
    ids := []string{"binance", "kraken", "bybit"}
    totals := map[string]float64{}

    for _, id := range ids {
        exchange := ccxt.CreateExchange(id, map[string]any{
            "apiKey": os.Getenv(strings.ToUpper(id) + "_APIKEY"),
            "secret": os.Getenv(strings.ToUpper(id) + "_SECRET"),
        })
        balance, err := exchange.FetchBalance()
        if err != nil {
            panic(err)
        }
        for currency, amount := range balance.Total {
            if amount != nil && *amount > 0 {
                totals[currency] += *amount
            }
        }
    }

    // tout valoriser en USDT à partir des tickers d'une seule plateforme
    pricing := ccxt.NewBinance(nil)
    tickers, err := pricing.FetchTickers()
    if err != nil {
        panic(err)
    }
    currencies := make([]string, 0, len(totals))
    for currency := range totals {
        currencies = append(currencies, currency)
    }
    sort.Strings(currencies)
    portfolio := 0.0
    for _, currency := range currencies {
        price := 0.0
        if currency == "USDT" {
            price = 1.0
        } else if ticker, ok := tickers.Tickers[currency+"/USDT"]; ok && ticker.Last != nil {
            price = *ticker.Last
        }
        if price == 0 {
            continue // pas de marché USDT — voir les notes ci-dessous
        }
        value := totals[currency] * price
        portfolio += value
        fmt.Printf("%-8s %14.6f ≈ %12.2f USDT\n", currency, totals[currency], value)
    }
    fmt.Printf("%-8s %14s ≈ %12.2f USDT\n", "TOTAL", "", portfolio)
}
```

```java tab="Java"
import io.github.ccxt.Exchange;

import java.util.*;

public class PortfolioTracker {

    @SuppressWarnings("unchecked")
    public static void main(String[] args) {
        var ids = List.of("binance", "kraken", "bybit");
        var totals = new TreeMap<String, Double>();

        // les instances dynamiques renvoient des maps non typées (voir sinon l'API typée par classe)
        for (var id : ids) {
            var config = new HashMap<String, Object>();
            config.put("apiKey", System.getenv(id.toUpperCase() + "_APIKEY"));
            config.put("secret", System.getenv(id.toUpperCase() + "_SECRET"));
            var exchange = Exchange.dynamicallyCreateInstance(id, config);
            var balance = (Map<String, Object>) exchange.fetchBalance().join();
            var total = (Map<String, Object>) balance.get("total");
            for (var entry : total.entrySet()) {
                var amount = (Number) entry.getValue();
                if (amount != null && amount.doubleValue() > 0) {
                    totals.merge(entry.getKey(), amount.doubleValue(), Double::sum);
                }
            }
        }

        // tout valoriser en USDT à partir des tickers d'une seule plateforme
        var pricing = new io.github.ccxt.exchanges.Binance();
        var tickers = pricing.fetchTickers();
        var portfolio = 0.0;
        for (var entry : totals.entrySet()) {
            Double price = null;
            if (entry.getKey().equals("USDT")) {
                price = 1.0;
            } else {
                var ticker = tickers.tickers.get(entry.getKey() + "/USDT");
                if (ticker != null) price = ticker.last;
            }
            if (price == null) continue;      // pas de marché USDT — voir les notes ci-dessous
            var value = entry.getValue() * price;
            portfolio += value;
            System.out.printf("%-8s %14.6f ≈ %12.2f USDT%n", entry.getKey(), entry.getValue(), value);
        }
        System.out.printf("%-8s %14s ≈ %12.2f USDT%n", "TOTAL", "", portfolio);
    }
}
```

## Les limites honnêtes d'un tracker de 40 lignes

- **`balance.total` est la vérité du portefeuille spot, pas votre patrimoine net.** La marge des
  dérivés, les soldes en staking/earn et les fonds engagés dans des ordres ouverts sont rapportés
  différemment selon la plateforme (`fetchBalance` accepte des params pour les autres types de
  compte — par exemple un `type` valant `'swap'` ou `'funding'` sur les exchanges qui séparent
  les portefeuilles). Bouclez sur les types de compte si vous les utilisez.
- **Le saut « pas de marché USDT ».** Les petits actifs ne se négocient parfois que contre BTC,
  ou pas du tout sur votre plateforme de valorisation. La version robuste se rabat sur
  `X/BTC` × `BTC/USDT`, ou valorise chaque solde sur l'exchange où il se trouve.
- **Les stablecoins à 1,0 sont une simplification** que nous n'avons faite que pour l'USDT
  lui-même. Pendant un depeg, votre tracker vous mentira sans sourciller — valorisez l'USDC et
  ses amis via leurs vrais marchés.
- **Mettez les tickers en cache.** Un seul `fetchTickers()` par exécution suffit ; des appels
  `fetchTicker` par devise sont le moyen le plus sûr de transformer un tracker en problème de
  rate limit.

De là, il n'y a qu'un pas vers une tâche cron qui ajoute une ligne CSV par heure — et vous voilà
soudain avec l'historique de portefeuille qu'aucun exchange ne vous donnera jamais à travers
plusieurs plateformes. Si vous le voulez en direct plutôt, `watchBalance` du
[billet sur les WebSockets](/fr/blog/realtime-market-data-with-websockets) pousse chaque changement au
moment où il se produit.
