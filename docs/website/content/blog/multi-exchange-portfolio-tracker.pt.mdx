Se você negocia em mais de uma exchange, conhece o ritual: faz login aqui, faz login ali, soma os
números de cabeça, esquece a poeira naquela terceira conta. A graça toda de uma API unificada
é que "quanto eu realmente tenho?" deveria ser um único script — chaves somente leitura, sem permissão
de ordens, nada para quebrar.

O plano: buscar os saldos de todas as exchanges, consolidar por moeda, precificar tudo em USDT
a partir dos tickers de uma única venue, imprimir uma tabela e um total.

```javascript tab="JavaScript"
import ccxt from 'ccxt';

const ids = ['binance', 'kraken', 'bybit'];
const totals = {};

for (const id of ids) {
    const exchange = new ccxt[id]({
        apiKey: process.env[`${id.toUpperCase()}_APIKEY`],
        secret: process.env[`${id.toUpperCase()}_SECRET`],
    });
    const balance = await exchange.fetchBalance();
    for (const [currency, amount] of Object.entries(balance.total)) {
        if (amount > 0) {
            totals[currency] = (totals[currency] ?? 0) + amount;
        }
    }
    await exchange.close();
}

// precifica tudo em USDT a partir dos tickers de uma única venue
const pricing = new ccxt.binance();
const tickers = await pricing.fetchTickers();
let portfolio = 0;
for (const [currency, amount] of Object.entries(totals)) {
    const price = (currency === 'USDT') ? 1 : tickers[`${currency}/USDT`]?.last;
    if (!price) continue;                    // sem mercado em USDT — veja as notas abaixo
    const value = amount * price;
    portfolio += value;
    console.log(`${currency.padEnd(8)} ${amount.toFixed(6).padStart(14)} ≈ ${value.toFixed(2).padStart(12)} USDT`);
}
console.log(`${'TOTAL'.padEnd(8)} ${' '.repeat(14)} ≈ ${portfolio.toFixed(2).padStart(12)} USDT`);
await pricing.close();
```

```go tab="Go"
package main

import (
    "fmt"
    "os"
    "sort"
    "strings"

    ccxt "github.com/ccxt/ccxt/go/v4"
)

func main() {
    ids := []string{"binance", "kraken", "bybit"}
    totals := map[string]float64{}

    for _, id := range ids {
        exchange := ccxt.CreateExchange(id, map[string]any{
            "apiKey": os.Getenv(strings.ToUpper(id) + "_APIKEY"),
            "secret": os.Getenv(strings.ToUpper(id) + "_SECRET"),
        })
        balance, err := exchange.FetchBalance()
        if err != nil {
            panic(err)
        }
        for currency, amount := range balance.Total {
            if amount != nil && *amount > 0 {
                totals[currency] += *amount
            }
        }
    }

    // precifica tudo em USDT a partir dos tickers de uma única venue
    pricing := ccxt.NewBinance(nil)
    tickers, err := pricing.FetchTickers()
    if err != nil {
        panic(err)
    }
    currencies := make([]string, 0, len(totals))
    for currency := range totals {
        currencies = append(currencies, currency)
    }
    sort.Strings(currencies)
    portfolio := 0.0
    for _, currency := range currencies {
        price := 0.0
        if currency == "USDT" {
            price = 1.0
        } else if ticker, ok := tickers.Tickers[currency+"/USDT"]; ok && ticker.Last != nil {
            price = *ticker.Last
        }
        if price == 0 {
            continue // sem mercado em USDT — veja as notas abaixo
        }
        value := totals[currency] * price
        portfolio += value
        fmt.Printf("%-8s %14.6f ≈ %12.2f USDT\n", currency, totals[currency], value)
    }
    fmt.Printf("%-8s %14s ≈ %12.2f USDT\n", "TOTAL", "", portfolio)
}
```

```java tab="Java"
import io.github.ccxt.Exchange;

import java.util.*;

public class PortfolioTracker {

    @SuppressWarnings("unchecked")
    public static void main(String[] args) {
        var ids = List.of("binance", "kraken", "bybit");
        var totals = new TreeMap<String, Double>();

        // instâncias dinâmicas retornam maps sem tipo (caso contrário, veja a API tipada por classe)
        for (var id : ids) {
            var config = new HashMap<String, Object>();
            config.put("apiKey", System.getenv(id.toUpperCase() + "_APIKEY"));
            config.put("secret", System.getenv(id.toUpperCase() + "_SECRET"));
            var exchange = Exchange.dynamicallyCreateInstance(id, config);
            var balance = (Map<String, Object>) exchange.fetchBalance().join();
            var total = (Map<String, Object>) balance.get("total");
            for (var entry : total.entrySet()) {
                var amount = (Number) entry.getValue();
                if (amount != null && amount.doubleValue() > 0) {
                    totals.merge(entry.getKey(), amount.doubleValue(), Double::sum);
                }
            }
        }

        // precifica tudo em USDT a partir dos tickers de uma única venue
        var pricing = new io.github.ccxt.exchanges.Binance();
        var tickers = pricing.fetchTickers();
        var portfolio = 0.0;
        for (var entry : totals.entrySet()) {
            Double price = null;
            if (entry.getKey().equals("USDT")) {
                price = 1.0;
            } else {
                var ticker = tickers.tickers.get(entry.getKey() + "/USDT");
                if (ticker != null) price = ticker.last;
            }
            if (price == null) continue;      // sem mercado em USDT — veja as notas abaixo
            var value = entry.getValue() * price;
            portfolio += value;
            System.out.printf("%-8s %14.6f ≈ %12.2f USDT%n", entry.getKey(), entry.getValue(), value);
        }
        System.out.printf("%-8s %14s ≈ %12.2f USDT%n", "TOTAL", "", portfolio);
    }
}
```

## Os limites honestos de um rastreador de 40 linhas

- **`balance.total` é a verdade da carteira spot, não o patrimônio líquido.** Margem de derivativos, saldos em
  staking/earn e fundos presos em ordens abertas são reportados de forma diferente em cada venue (`fetchBalance`
  aceita params para outros tipos de conta — por exemplo, um `type` igual a `'swap'` ou `'funding'` em
  exchanges que separam as carteiras). Itere sobre os tipos de conta se você os usa.
- **O pulo do "sem mercado em USDT".** Ativos pequenos podem ser negociados apenas contra BTC, ou nem isso, na
  sua venue de precificação. A versão robusta recorre a `X/BTC` × `BTC/USDT`, ou precifica cada
  saldo na exchange onde ele está.
- **Stablecoins a 1,0 é uma simplificação** que fizemos apenas para o próprio USDT. Durante um depeg,
  seu rastreador vai mentir para você sem o menor pudor — precifique USDC e companhia pelos seus mercados reais.
- **Faça cache dos tickers.** Um `fetchTickers()` por execução é ótimo; chamadas `fetchTicker` por moeda
  são a forma como um rastreador vira um problema de rate limit.

Daqui é um passo curto até um cron job que acrescenta uma linha de CSV por hora — e de repente você
tem o gráfico de histórico de portfólio que nenhuma exchange jamais vai lhe dar entre venues. Se preferir
ao vivo, o `watchBalance` do
[post sobre WebSocket](/pt/blog/realtime-market-data-with-websockets) empurra cada mudança assim que
ela acontece.
