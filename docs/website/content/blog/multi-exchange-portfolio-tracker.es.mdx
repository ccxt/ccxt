Si operas en más de un exchange, conoces el ritual: inicia sesión aquí, inicia sesión allá, suma
números de cabeza, olvida el polvo que quedó en esa tercera cuenta. La gracia de una API unificada
es que "¿cuánto tengo en realidad?" debería ser un solo script — claves de solo lectura, sin
permisos de órdenes, nada que romper.

El plan: obtener los balances de cada exchange, combinarlos por moneda, valorar todo en USDT con
los tickers de un solo venue, imprimir una tabla y un total.

```javascript tab="JavaScript"
import ccxt from 'ccxt';

const ids = ['binance', 'kraken', 'bybit'];
const totals = {};

for (const id of ids) {
    const exchange = new ccxt[id]({
        apiKey: process.env[`${id.toUpperCase()}_APIKEY`],
        secret: process.env[`${id.toUpperCase()}_SECRET`],
    });
    const balance = await exchange.fetchBalance();
    for (const [currency, amount] of Object.entries(balance.total)) {
        if (amount > 0) {
            totals[currency] = (totals[currency] ?? 0) + amount;
        }
    }
    await exchange.close();
}

// valorar todo en USDT con los tickers de un solo venue
const pricing = new ccxt.binance();
const tickers = await pricing.fetchTickers();
let portfolio = 0;
for (const [currency, amount] of Object.entries(totals)) {
    const price = (currency === 'USDT') ? 1 : tickers[`${currency}/USDT`]?.last;
    if (!price) continue;                    // sin mercado en USDT — ver notas más abajo
    const value = amount * price;
    portfolio += value;
    console.log(`${currency.padEnd(8)} ${amount.toFixed(6).padStart(14)} ≈ ${value.toFixed(2).padStart(12)} USDT`);
}
console.log(`${'TOTAL'.padEnd(8)} ${' '.repeat(14)} ≈ ${portfolio.toFixed(2).padStart(12)} USDT`);
await pricing.close();
```

```go tab="Go"
package main

import (
    "fmt"
    "os"
    "sort"
    "strings"

    ccxt "github.com/ccxt/ccxt/go/v4"
)

func main() {
    ids := []string{"binance", "kraken", "bybit"}
    totals := map[string]float64{}

    for _, id := range ids {
        exchange := ccxt.CreateExchange(id, map[string]any{
            "apiKey": os.Getenv(strings.ToUpper(id) + "_APIKEY"),
            "secret": os.Getenv(strings.ToUpper(id) + "_SECRET"),
        })
        balance, err := exchange.FetchBalance()
        if err != nil {
            panic(err)
        }
        for currency, amount := range balance.Total {
            if amount != nil && *amount > 0 {
                totals[currency] += *amount
            }
        }
    }

    // valorar todo en USDT con los tickers de un solo venue
    pricing := ccxt.NewBinance(nil)
    tickers, err := pricing.FetchTickers()
    if err != nil {
        panic(err)
    }
    currencies := make([]string, 0, len(totals))
    for currency := range totals {
        currencies = append(currencies, currency)
    }
    sort.Strings(currencies)
    portfolio := 0.0
    for _, currency := range currencies {
        price := 0.0
        if currency == "USDT" {
            price = 1.0
        } else if ticker, ok := tickers.Tickers[currency+"/USDT"]; ok && ticker.Last != nil {
            price = *ticker.Last
        }
        if price == 0 {
            continue // sin mercado en USDT — ver notas más abajo
        }
        value := totals[currency] * price
        portfolio += value
        fmt.Printf("%-8s %14.6f ≈ %12.2f USDT\n", currency, totals[currency], value)
    }
    fmt.Printf("%-8s %14s ≈ %12.2f USDT\n", "TOTAL", "", portfolio)
}
```

```java tab="Java"
import io.github.ccxt.Exchange;

import java.util.*;

public class PortfolioTracker {

    @SuppressWarnings("unchecked")
    public static void main(String[] args) {
        var ids = List.of("binance", "kraken", "bybit");
        var totals = new TreeMap<String, Double>();

        // las instancias dinámicas devuelven mapas sin tipar (si no, usa la API tipada por clase)
        for (var id : ids) {
            var config = new HashMap<String, Object>();
            config.put("apiKey", System.getenv(id.toUpperCase() + "_APIKEY"));
            config.put("secret", System.getenv(id.toUpperCase() + "_SECRET"));
            var exchange = Exchange.dynamicallyCreateInstance(id, config);
            var balance = (Map<String, Object>) exchange.fetchBalance().join();
            var total = (Map<String, Object>) balance.get("total");
            for (var entry : total.entrySet()) {
                var amount = (Number) entry.getValue();
                if (amount != null && amount.doubleValue() > 0) {
                    totals.merge(entry.getKey(), amount.doubleValue(), Double::sum);
                }
            }
        }

        // valorar todo en USDT con los tickers de un solo venue
        var pricing = new io.github.ccxt.exchanges.Binance();
        var tickers = pricing.fetchTickers();
        var portfolio = 0.0;
        for (var entry : totals.entrySet()) {
            Double price = null;
            if (entry.getKey().equals("USDT")) {
                price = 1.0;
            } else {
                var ticker = tickers.tickers.get(entry.getKey() + "/USDT");
                if (ticker != null) price = ticker.last;
            }
            if (price == null) continue;      // sin mercado en USDT — ver notas más abajo
            var value = entry.getValue() * price;
            portfolio += value;
            System.out.printf("%-8s %14.6f ≈ %12.2f USDT%n", entry.getKey(), entry.getValue(), value);
        }
        System.out.printf("%-8s %14s ≈ %12.2f USDT%n", "TOTAL", "", portfolio);
    }
}
```

## Los límites honestos de un rastreador de 40 líneas

- **`balance.total` es la verdad de la billetera spot, no el patrimonio neto.** El margen en
  derivados, los balances en staking/earn y los fondos en órdenes abiertas se reportan de forma
  distinta según el venue (`fetchBalance` acepta params para otros tipos de cuenta — por ejemplo
  un `type` de `'swap'` o `'funding'` en exchanges que separan las billeteras). Recorre los tipos
  de cuenta si los usas.
- **El salto por "sin mercado en USDT".** Los activos pequeños puede que solo coticen contra BTC
  o directamente no coticen en tu venue de valoración. La versión robusta recurre a `X/BTC` ×
  `BTC/USDT`, o valora cada balance en el exchange donde vive.
- **Las stablecoins a 1.0 es una simplificación** que hicimos solo para el propio USDT. Durante un
  depeg, tu rastreador te mentirá con toda tranquilidad — valora USDC y compañía a través de sus
  mercados reales.
- **Cachea los tickers.** Un `fetchTickers()` por ejecución está bien; las llamadas a
  `fetchTicker` por moneda son la forma en que un rastreador se convierte en un problema de rate
  limit.

Desde aquí hay un pequeño paso hasta un cron job que añade una fila CSV cada hora — y de repente
tienes el gráfico de historial de portafolio que ningún exchange te dará jamás a través de varios
venues. Si lo quieres en vivo, `watchBalance` del
[post sobre WebSockets](/es/blog/realtime-market-data-with-websockets) te envía cada cambio en el
momento en que ocurre.
